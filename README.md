function productForm(id){let p=db.products.find(p=>p.id===id)||{id:'p'+Date.now(),name:'',price:0,desc:'',img:'',codes:[]};modal(`<h2>${id?'แก้ไขสินค้า':'เพิ่มสินค้า'}</h2><div class="field"><label>ชื่อสินค้า</label><input id="pn" value="${esc(p.name)}"></div><div class="row"><div class="field"><label>ราคา (บาท)</label><input id="pp" type="number" min="0" value="${p.price}"></div><div class="field"><label>สต็อกโค้ด / บัญชี (หนึ่งรายการต่อบรรทัด)</label><textarea id="pcodes" placeholder="CODE-001">${p.codes.map(esc).join('\n')}</textarea></div></div><div class="field"><label>รายละเอียด</label><textarea id="pd">${esc(p.desc)}</textarea></div><div class="field"><label>รูปสินค้า (เลือกจากมือถือ)</label><input id="pimg" type="file" accept="image/*" onchange="previewImage()"><div id="imgpreview" class="mini">${p.img?'มีรูปเดิมแล้ว':'ยังไม่มีรูป'}</div></div><div class="notice">ใช้รหัสสินค้าทดสอบเท่านั้น ห้ามใส่ข้อมูลลูกค้าหรือรหัสผ่านจริง การส่งมอบอัตโนมัติยังไม่เปิดใช้งานในเดโมนี้</div><div class="actions"><button class="secondary" onclick="closeModal()">ยกเลิก</button><button onclick="saveProduct('${p.id}',${!!id})">บันทึกสินค้า</button></div>`);window.pendingImage=p.img||''}
function previewImage(){let f=$('#pimg').files[0];if(!f)return;if(f.size>1200000)return alert('กรุณาใช้รูปขนาดไม่เกิน 1.2 MB');let r=new FileReader();r.onload=()=>{window.pendingImage=r.result;$('#imgpreview').textContent='เลือกรูปแล้ว'};r.readAsDataURL(f)}
function saveProduct(id,editing){let name=$('#pn').value.trim(),price=Number($('#pp').value),codes=$('#pcodes').value.split(/\n/).map(x=>x.trim()).filter(Boolean);if(!name||price<0||!Number.isFinite(price))return alert('ตรวจสอบชื่อสินค้าและราคา');let p={id,name,price,desc:$('#pd').value,img:window.pendingImage||'',codes};if(editing)db.products=db.products.map(x=>x.id===id?p:x);else db.products.unshift(p);save();closeModal();render();adminTab('products',document.querySelector('.tab'))}
function deleteProduct(id){if(confirm('ยืนยันลบสินค้านี้?')){db.products=db.products.filter(p=>p.id!==id);save();render();adminTab('products',document.querySelector('.tab'))}}
function setStatus(id,status){let o=db.orders.find(o=>o.id===id);if(o){o.status=status;save()}}
render();
